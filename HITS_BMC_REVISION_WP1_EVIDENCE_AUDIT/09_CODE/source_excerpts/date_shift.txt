0058: parse_datetime <- function(x) {
0059:   x <- trimws(as.character(x))
0060:   x[x %in% c("", "NA", "N/A", "NULL")] <- NA_character_
0061:   out <- as.POSIXct(rep(NA_character_, length(x)), tz = "Asia/Shanghai")
0062:
0063:   numeric_x <- suppressWarnings(as.numeric(x))
0064:   excel <- !is.na(numeric_x) & numeric_x > 20000 & numeric_x < 70000
0065:   out[excel] <- as.POSIXct(as.Date(numeric_x[excel], origin = "1899-12-30"), tz = "Asia/Shanghai")
0066:
0067:   formats <- c(
0068:     "%Y-%m-%d %H:%M:%S", "%Y-%m-%d %H:%M", "%Y-%m-%d",
0069:     "%Y/%m/%d %H:%M:%S", "%Y/%m/%d %H:%M", "%Y/%m/%d",
0070:     "%Y.%m.%d %H:%M:%S", "%Y.%m.%d"
0071:   )
0072:   remaining <- is.na(out) & !is.na(x)
0073:   for (fmt in formats) {
0074:     if (!any(remaining)) break
0075:     parsed <- as.POSIXct(x[remaining], format = fmt, tz = "Asia/Shanghai")
0076:     idx <- which(remaining)[!is.na(parsed)]
0077:     out[idx] <- parsed[!is.na(parsed)]
0078:     remaining <- is.na(out) & !is.na(x)
0079:   }
0080:   out
0081: }

0088: pseudonymize <- function(d, label) {
0089:   original_n <- nrow(d)
0090:   patient_source <- if ("patient_sn" %in% names(d)) trimws(d$patient_sn) else rep(NA_character_, nrow(d))
0091:   pseudo_patient <- unname(patient_map[patient_source])
0092:

0094:   if (length(drop_cols)) d[drop_cols] <- NULL
0095:
0096:   if ("patient_sn" %in% names(d)) {
0097:     d$patient_sn <- pseudo_patient
0098:     names(d)[names(d) == "patient_sn"] <- "research_patient_id"
0099:   }
0100:
0101:   for (col in intersect(admission_columns, names(d))) {
0102:     d[[col]] <- unname(admission_map[trimws(d[[col]])])
0103:   }
0104:   names(d)[names(d) == "基线_inpatient_no_id"] <- "baseline_admission_id_pseudo"
0105:   names(d)[names(d) == "inpatient_no_id"] <- "admission_id_pseudo"
0106:

0113:     nonmissing_raw <- !is.na(raw) & nzchar(trimws(raw))
0114:     unparsed_n <- sum(nonmissing_raw & is.na(parsed))
0115:     d[[col]] <- ifelse(is.na(shifted), NA_character_, format(shifted, "%Y-%m-%d %H:%M:%S", tz = "Asia/Shanghai"))

0117:       dataset = label,
0118:       field = col,
0119:       nonmissing_source = sum(nonmissing_raw),
0120:       parsed_and_shifted = sum(!is.na(shifted)),
0121:       unparsed_redacted = unparsed_n,
0122:       stringsAsFactors = FALSE
0123:     )
0124:   }
0125:
0126:   stopifnot(nrow(d) == original_n)
