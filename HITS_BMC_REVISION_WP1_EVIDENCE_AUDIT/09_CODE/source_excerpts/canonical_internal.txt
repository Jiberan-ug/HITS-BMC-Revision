0171:         if not self.impute and not np.isfinite(raw).all():

0327:     effects: list[float] = []
0328:     for i, variable in enumerate(features):
0329:         increment = INCREMENTS[variable]
0330:         median = prep.medians_raw_[i]
0331:         if variable in LOG_VARS:

0333:             standardised_increment = raw_delta / prep.scale_[i]

0515: def calibration_stats(y: np.ndarray, probability: np.ndarray) -> tuple[float, float]:
0516:     if len(y) < 20 or len(np.unique(y)) < 2:
0517:         return np.nan, np.nan
0518:     probability = np.clip(np.asarray(probability, dtype=float), 1e-6, 1 - 1e-6)
0519:     logit = np.log(probability / (1 - probability))
0520:     try:
0521:         model = sm.GLM(y, sm.add_constant(logit), family=sm.families.Binomial()).fit()
0522:         return safe_float(model.params[0]), safe_float(model.params[1])
0523:     except Exception:
0524:         return np.nan, np.nan

0528:     if len(np.unique(y)) < 2:
0529:         return np.nan, np.nan, 0
0530:     rng = np.random.default_rng(stable_seed("metric-ci:" + label))
0531:     values: list[float] = []
0532:     for _ in range(n_boot):
0533:         idx = rng.integers(0, len(y), len(y))
0534:         if len(np.unique(y[idx])) == 2:
0535:             values.append(safe_auc(y[idx], score[idx]))

0577: def paired_delta(
0578:     records_a: pd.DataFrame,
0579:     records_b: pd.DataFrame,
0580:     label: str,
0581:     n_boot: int,
0582: ) -> dict[str, Any]:
0583:     a = patient_mean_predictions(records_a)[["research_patient_id", "y", "prediction"]].rename(columns={"prediction": "a"})
0584:     b = patient_mean_predictions(records_b)[["research_patient_id", "y", "prediction"]].rename(columns={"prediction": "b"})
0585:     merged = a.merge(b, on=["research_patient_id", "y"], how="inner", validate="one_to_one")
0586:     y = merged.y.to_numpy(dtype=int)
0587:     p_a = merged.a.to_numpy(dtype=float)
0588:     p_b = merged.b.to_numpy(dtype=float)
0589:     point = safe_auc(y, p_a) - safe_auc(y, p_b)
0590:     brier_delta = safe_float(brier_score_loss(y, p_a) - brier_score_loss(y, p_b))
0591:     rng = np.random.default_rng(stable_seed("paired:" + label))
0592:     values: list[float] = []
0593:     brier_values: list[float] = []
0594:     for _ in range(n_boot):
0595:         idx = rng.integers(0, len(merged), len(merged))
0596:         if len(np.unique(y[idx])) != 2:
0597:             continue
0598:         values.append(safe_auc(y[idx], p_a[idx]) - safe_auc(y[idx], p_b[idx]))
0599:         brier_values.append(safe_float(brier_score_loss(y[idx], p_a[idx]) - brier_score_loss(y[idx], p_b[idx])))
0600:     return {
0601:         "comparison": label,
0602:         "n_common": int(len(merged)),
0603:         "auc_model_a": safe_auc(y, p_a),
0604:         "auc_model_b": safe_auc(y, p_b),
0605:         "delta_auc_a_minus_b": safe_float(point),
0606:         "delta_auc_ci_lower": safe_float(np.quantile(values, 0.025)) if values else np.nan,
0607:         "delta_auc_ci_upper": safe_float(np.quantile(values, 0.975)) if values else np.nan,
0608:         "delta_auc_bootstrap_valid": len(values),
0609:         "brier_delta_a_minus_b": brier_delta,
0610:         "brier_delta_ci_lower": safe_float(np.quantile(brier_values, 0.025)) if brier_values else np.nan,
0611:         "brier_delta_ci_upper": safe_float(np.quantile(brier_values, 0.975)) if brier_values else np.nan,
0612:         "ci_supports_a_higher_auc": bool(values and np.quantile(values, 0.025) > 0),
0613:         "ci_supports_a_lower_auc": bool(values and np.quantile(values, 0.975) < 0),

0636:     work = frame.reset_index(drop=True)

0641:     rng = np.random.default_rng(stable_seed("development-bootstrap:" + label))
